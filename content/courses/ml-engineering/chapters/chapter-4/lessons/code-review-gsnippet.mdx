def caesar_encode(message, offset):
    """Encode a message using a Caesar cipher with given offset."""
    result = []
    for ch in message:
        if ch.isalpha():
            base = ord('a') if ch.islower() else ord('A')
            shifted = (ord(ch) - base + offset) % 26 + base
            result.append(chr(shifted))
        else:
            result.append(ch)
    return ''.join(result)

def caesar_decode(message, offset):
    """Decode a message encoded with a Caesar cipher using given offset."""
    return caesar_encode(message, -offset)
